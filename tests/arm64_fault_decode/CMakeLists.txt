# arm64_fault_decode: the AArch64 MMIO fault decoder/emulator
# (third_party/xboxrecomp/src/platform/arm64_fault_decode.c).
#
# Native (any host; the decoder has no OS dependencies):
#   cmake -G Ninja -S tests/arm64_fault_decode -B build-a64dec
#   cmake --build build-a64dec && ctest --test-dir build-a64dec --output-on-failure
#     -> arm64-fault-decode (vectors + semantics)
#
# AArch64 (Quest, or this host via the QEMU executor):
#   cmake -G Ninja -S tests/arm64_fault_decode -B build-a64dec-arm64 \
#         -DCMAKE_TOOLCHAIN_FILE=/root/xtool/android-arm64-static.cmake
#   cmake --build build-a64dec-arm64 && ctest --test-dir build-a64dec-arm64 --output-on-failure
#     -> arm64-fault-decode (same, as an arm64 binary)
#     -> arm64-fault-e2e    (real SIGSEGV on a PROT_NONE aperture -> resume)
#
# vectors_gen.h is generated by gen_vectors.py from the NDK's clang and
# llvm-objdump and is checked in; the `regen-vectors` target reruns it when
# the NDK is present.
cmake_minimum_required(VERSION 3.20)
project(arm64_fault_decode_test C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(PLATFORM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/xboxrecomp/src/platform")

add_library(arm64_fault_decode STATIC "${PLATFORM_DIR}/arm64_fault_decode.c")
target_include_directories(arm64_fault_decode PUBLIC "${PLATFORM_DIR}")
if(MSVC)
    target_compile_options(arm64_fault_decode PRIVATE /W4)
else()
    target_compile_options(arm64_fault_decode PRIVATE -Wall -Wextra -Werror)
endif()

enable_testing()

add_executable(arm64-fault-decode-test test_decode.c)
target_link_libraries(arm64-fault-decode-test PRIVATE arm64_fault_decode)
target_include_directories(arm64-fault-decode-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME arm64-fault-decode COMMAND arm64-fault-decode-test)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$" OR ANDROID_ABI STREQUAL "arm64-v8a")
    add_executable(arm64-fault-e2e-test test_fault_e2e.c)
    target_link_libraries(arm64-fault-e2e-test PRIVATE arm64_fault_decode)
    # LSE for the atomics cases (Quest 3 is Armv8.2).
    target_compile_options(arm64-fault-e2e-test PRIVATE -march=armv8.2-a -std=gnu11)
    add_test(NAME arm64-fault-e2e COMMAND arm64-fault-e2e-test)
    set_tests_properties(arm64-fault-e2e PROPERTIES TIMEOUT 60)
endif()

find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_custom_target(regen-vectors
        COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/gen_vectors.py"
                --out "${CMAKE_CURRENT_SOURCE_DIR}/vectors_gen.h"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        COMMENT "Regenerating arm64 decode vectors with the NDK")
endif()
